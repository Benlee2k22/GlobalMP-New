cmake_minimum_required(VERSION 3.20)
project(GlobalMPRelay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
if(NOT DEFINED ETS2MP_SOURCE)
 message(FATAL_ERROR "Set -DETS2MP_SOURCE=/path/to/ETS2MP")
endif()
set(RAKNET_DIR "${ETS2MP_SOURCE}/src/Vendor/RakNet/Source")
file(GLOB RAKNET_SOURCES "${RAKNET_DIR}/*.cpp")
list(FILTER RAKNET_SOURCES EXCLUDE REGEX ".*/(HTTPConnection2|Router2|UDPForwarder)[.]cpp$")
add_library(raknet STATIC ${RAKNET_SOURCES})
target_include_directories(raknet PUBLIC "${RAKNET_DIR}")
if(NOT MSVC)
 target_compile_options(raknet PRIVATE -Wno-narrowing)
 target_compile_definitions(raknet PRIVATE INVALID_SOCKET=-1)
 target_link_libraries(raknet PUBLIC pthread)
endif()
add_executable(globalmp-relay main.cpp)
target_link_libraries(globalmp-relay PRIVATE raknet)
